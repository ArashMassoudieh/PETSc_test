cmake_minimum_required(VERSION 3.16)
project(btc_averaging LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(UTILITIES_DIR "${REPO_ROOT}/Utilities" CACHE PATH "Utilities source checkout")
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(GSL REQUIRED)
find_package(Armadillo REQUIRED)
if(NOT (VTK_INCLUDE_DIR AND VTK_LIBRARIES))
  find_package(VTK REQUIRED COMPONENTS CommonCore CommonDataModel CommonExecutionModel IOXML)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFTW REQUIRED IMPORTED_TARGET fftw3)
# Explicit paths also support PETSc installations without a pkg-config file.
if(PETSC_INCLUDE_DIR AND PETSC_LIBRARY)
  add_library(petsc_dependency INTERFACE)
  target_include_directories(petsc_dependency INTERFACE "${PETSC_INCLUDE_DIR}")
  target_link_libraries(petsc_dependency INTERFACE "${PETSC_LIBRARY}")
else()
  pkg_search_module(PETSC REQUIRED IMPORTED_TARGET PETSc petsc)
  add_library(petsc_dependency INTERFACE)
  target_link_libraries(petsc_dependency INTERFACE PkgConfig::PETSC)
endif()
set(CORE Particle.cpp Pathway.cpp PathwaySet.cpp copula_analysis.cpp grid.cpp
    petscmatrix.cpp petscsolver.cpp petscvector.cpp sim_helpers.cpp)
list(TRANSFORM CORE PREPEND "${REPO_ROOT}/")
set(UTIL Matrix.cpp Matrix_arma.cpp NormalDist.cpp QuickSort.cpp Utilities.cpp Vector.cpp Vector_arma.cpp)
list(TRANSFORM UTIL PREPEND "${UTILITIES_DIR}/")
add_executable(ensemble_probe ensemble_probe.cpp ${CORE} ${UTIL})
target_include_directories(ensemble_probe PRIVATE "${REPO_ROOT}" "${UTILITIES_DIR}" ${ARMADILLO_INCLUDE_DIRS} ${VTK_INCLUDE_DIR})
target_compile_definitions(ensemble_probe PRIVATE GSL Arash GRID_USE_VTK)
target_compile_options(ensemble_probe PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/compat.h")
target_link_libraries(ensemble_probe PRIVATE petsc_dependency MPI::MPI_CXX GSL::gsl GSL::gslcblas PkgConfig::FFTW ${ARMADILLO_LIBRARIES} ${VTK_LIBRARIES})
if(COMMAND vtk_module_autoinit)
  vtk_module_autoinit(TARGETS ensemble_probe MODULES ${VTK_LIBRARIES})
endif()
